A.1　Rego策略语言标准示例
该规则中定义了一个tee设备带有两个异构TEE外设，并且match_node只有当tee的virtcca_rim为123，一个异构TEE外设的ascend_gpu_validated=true，另一个异构TEE外设的nvidia_gpu_validated=true时才为true，即满足整个策略。
package trustnode  
default match_node := false
match_node {
    allowed_node(input.trusted_node)
    every d in input.trusted_node. trusted_hetero_devices {
         allowed_device(d)
    }
 }
allowed_node(node) {
     node.tee.virtcca_rim == "123" 
} 
allowed_device(device) {
      device.ascend_gpu_validated == true 
}  
allowed_device(device) {
     device.nvidia_gpu_validated == true 
}
